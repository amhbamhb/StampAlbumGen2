#------------------------------------------

ROW_START_FS (HN 6 0.1 6.0)
STAMP_ADD (26.0 22.0 "1/2d \n green"             ""  "2" "")
ROW_START_FS (HN 6 0.1 6.0)

ROW_START_FS (HN 6 0.1 6.0)
